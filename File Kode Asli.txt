public function show(Letter $letter)
    {
        // Pastikan path benar
        if (!Storage::exists($letter->file_path)) {
            return response()->json(['error' => 'File tidak ditemukan'], 404);
        }
        
        $path = Storage::path($letter->file_path);
        $mimeType = Storage::mimeType($letter->file_path);

        // Kirim file ke browser untuk ditampilkan
        return response()->file($path, [
            'Content-Type' => $mimeType,
            'Content-Disposition' => 'inline; filename="' . $letter->original_file_name . '"'
        ]);
    }
